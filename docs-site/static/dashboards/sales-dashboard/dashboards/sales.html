SELECT month, channel, sum(revenue) as revenue
FROM "sales_dashboard"."main"."fct_sales"
GROUP BY 1, 2

Regional Revenue Share

Compiled SQL 
SELECT region, sum(revenue) as revenue
FROM "sales_dashboard"."main"."fct_sales"
GROUP BY 1
