FROM "product_analytics"."main"."fct_product_usage"

Activation by Plan compiled 
SELECT week, plan, sum(activated_users) as activated_users
FROM "product_analytics"."main"."fct_product_usage"
GROUP BY 1, 2
Activation Rate by Plan compiled 
SELECT
  week,
  plan,
  round(sum(activated_users) * 100.0 / nullif(sum(active_users), 0), 1) as activation_rate
FROM "product_analytics"."main"."fct_product_usage"
GROUP BY 1, 2
Sessions per Active User compiled 
SELECT
  week,
  round(sum(sessions) * 1.0 / nullif(sum(active_users), 0), 2) as sessions_per_user
FROM "product_analytics"."main"."fct_product_usage"
GROUP BY 1
Sessions by Plan compiled 
SELECT
  plan,
  sum(sessions) as sessions
FROM "product_analytics"."main"."fct_product_usage"
GROUP BY 1
ORDER BY sessions DESC
Most active accounts compiled
