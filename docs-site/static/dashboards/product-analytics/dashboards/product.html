Weekly active users compiled 
WITH weekly AS (
  SELECT week, sum(active_users) AS active_users
  FROM "product_analytics"."main"."fct_product_usage"
  GROUP BY 1
)
SELECT active_users, lag(active_users) OVER (ORDER BY week) AS previous
FROM weekly
ORDER BY week DESC
LIMIT 1
Weekly sessions compiled 
WITH weekly AS (
  SELECT week, sum(sessions) AS sessions
  FROM "product_analytics"."main"."fct_product_usage"
  GROUP BY 1
)
SELECT sessions, lag(sessions) OVER (ORDER BY week) AS previous
FROM weekly
ORDER BY week DESC
LIMIT 1

  week,
  plan,
  round(sum(activated_users) * 100.0 / nullif(sum(active_users), 0), 1) as activation_rate
FROM "product_analytics"."main"."fct_product_usage"
GROUP BY 1, 2
