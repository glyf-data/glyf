Weekly active users compiled 
WITH weekly AS (
  SELECT week, sum(active_users) AS active_users
  FROM "product_analytics"."main"."fct_product_usage"
  GROUP BY 1

FROM weekly
ORDER BY week DESC, active_users, previous

From Sign-up to Power Use compiled 
SELECT 1 AS step_order, 'All accounts' AS step, count(*) AS accounts
FROM "product_analytics"."main"."fct_account_sessions"
UNION ALL
SELECT 2, 'Regular use', count(*)
FROM "product_analytics"."main"."fct_account_sessions"
WHERE sessions >= 15
UNION ALL
SELECT 3, 'Heavy use', count(*)
FROM "product_analytics"."main"."fct_account_sessions"
WHERE sessions >= 40
UNION ALL
SELECT 4, 'Heavy use, long sessions', count(*)
FROM "product_analytics"."main"."fct_account_sessions"
WHERE sessions >= 40 AND avg_session_minutes >= 15
ORDER BY step_order
How Accounts Use the Product compiled 
WITH accounts AS (
  SELECT
    plan,
    CASE
      WHEN sessions >= 40 THEN 'Heavy use'
      WHEN sessions >= 15 THEN 'Regular use'
      ELSE 'Light use'
    END AS activity,
    CASE WHEN avg_session_minutes >= 15 THEN 'Long sessions' ELSE 'Short sessions' END AS session_length
  FROM "product_analytics"."main"."fct_account_sessions"
)
SELECT plan AS stage, activity AS next_stage, count(*) AS accounts
FROM accounts
GROUP BY 1, 2
UNION ALL
SELECT activity, session_length, count(*)
FROM accounts
GROUP BY 1, 2
ORDER BY 1, 2
